-- 创建数据库
CREATE DATABASE IF NOT EXISTS finance_recon CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE finance_recon;

-- 支付渠道流水表
DROP TABLE IF EXISTS dwd_pay_channel_flow;
CREATE TABLE dwd_pay_channel_flow (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    flow_id VARCHAR(64) NOT NULL,
    order_no VARCHAR(64) NOT NULL,
    channel_code VARCHAR(32) NOT NULL,
    trans_amount DECIMAL(18, 4) NOT NULL, -- 核心：定点数
    trans_time DATETIME(3) NOT NULL,      -- 支持毫秒
    status VARCHAR(20) NOT NULL,
    INDEX idx_order_no (order_no),        -- 关键索引：Join 依赖
    INDEX idx_dt_status (trans_time, status) -- 辅助索引：过滤依赖
) ENGINE=InnoDB;

-- 核心账务表
DROP TABLE IF EXISTS dwd_core_account_log;
CREATE TABLE dwd_core_account_log (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    log_id VARCHAR(64) NOT NULL,
    order_no VARCHAR(64) NOT NULL,
    account_amount DECIMAL(18, 4) NOT NULL,
    book_time DATETIME(3) NOT NULL,
    status TINYINT NOT NULL,              -- 1:成功, 0:失败
    INDEX idx_order_no (order_no),
    INDEX idx_dt_status (book_time, status)
) ENGINE=InnoDB;

CREATE PROCEDURE generate_realistic_trap_data()
BEGIN
    -- 1. Declare variables (Must be first)
    DECLARE v_total_normal_rows INT DEFAULT 2000000; 
    DECLARE v_batch_size INT DEFAULT 50000;          
    DECLARE v_batch_count INT;
    DECLARE v_current_batch INT;
    
    -- Error handler
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
        SELECT 'Error occurred, transaction rolled back.' AS Result;
    END;

    -- 2. Initialize variables
    SET @i = 0;
    SET @j = 0;
    SET @row_num = 0;
    SET @log_num = 0;
    SET @current_batch = 0;
    
    SET v_batch_count = CEIL(v_total_normal_rows / v_batch_size);
    SET v_current_batch = 0;

    -- 3. Part 1: Trap Data
    START TRANSACTION;

    -- Trap A: Float Precision
    INSERT INTO dwd_pay_channel_flow (flow_id, order_no, channel_code, trans_amount, trans_time, status)
    VALUES ('F_ERR_FLOAT', 'O_ERR_FLOAT', 'WECHAT', CAST('0.30000000000000004' AS DECIMAL(18,4)), '2026-03-13 12:00:00.123', 'SUCCESS');
    INSERT INTO dwd_core_account_log (log_id, order_no, account_amount, book_time, status)
    VALUES ('L_ERR_FLOAT', 'O_ERR_FLOAT', 0.30, '2026-03-13 12:00:05.000', 1);

    -- Trap B: Duplicate Orders
    INSERT INTO dwd_pay_channel_flow (flow_id, order_no, channel_code, trans_amount, trans_time, status)
    VALUES 
    ('F_DUP_01', 'O_DUP_ORDER', 'UNIONPAY', 50.00, '2026-03-13 14:00:00.123', 'SUCCESS'),
    ('F_DUP_02', 'O_DUP_ORDER', 'UNIONPAY', 50.00, '2026-03-13 14:00:00.133', 'SUCCESS');
    INSERT INTO dwd_core_account_log (log_id, order_no, account_amount, book_time, status)
    VALUES ('L_DUP_01', 'O_DUP_ORDER', 50.00, '2026-03-13 14:00:01.000', 1);

    -- Trap C: Data Skew (Pay Table: 100,000 rows)
    INSERT INTO dwd_pay_channel_flow (flow_id, order_no, channel_code, trans_amount, trans_time, status)
    SELECT 
        CONCAT('F_SKEW_', LPAD(@i := @i + 1, 6, '0')), 
        'O_SKEW_HOT', 
        'SKEW_CHANNEL', 
        10.00, 
        DATE_ADD(NOW(), INTERVAL FLOOR(RAND() * 30) DAY), 
        'SUCCESS'
    FROM (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) a,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) b,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) c,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) d,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) e,
         (SELECT @i := 0) r
    LIMIT 100000; 

    -- Trap C: Data Skew (Core Table: 99,999 rows - Intentional Mismatch)
    INSERT INTO dwd_core_account_log (log_id, order_no, account_amount, book_time, status)
    SELECT 
        CONCAT('L_SKEW_', LPAD(@j := @j + 1, 6, '0')), 
        'O_SKEW_HOT', 
        10.00, 
        DATE_ADD(NOW(), INTERVAL FLOOR(RAND() * 30) DAY), 
        1
    FROM (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) a,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) b,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) c,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) d,
         (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) e,
         (SELECT @j := 0) r
    LIMIT 99999; 

    COMMIT; 

    -- 4. Part 2: Large Volume Random Normal Data
    WHILE v_current_batch < v_batch_count DO
        
        START TRANSACTION;
        
        SET @row_num = v_current_batch * v_batch_size;
        
        -- Insert Pay Channel Flow
        INSERT INTO dwd_pay_channel_flow (flow_id, order_no, channel_code, trans_amount, trans_time, status)
        SELECT 
            CONCAT('F_NORM_', LPAD(@row_num := @row_num + 1, 10, '0')),

                WHEN RAND() > 0.9 THEN ROUND(RAND() * 10000, 2) 
                ELSE ROUND(RAND() * 500, 2)                     
            END,
            DATE_SUB(NOW(), INTERVAL FLOOR(RAND() * 30 * 24 * 60) MINUTE), 
            ELT(FLOOR(1 + RAND() * 10), 'SUCCESS', 'SUCCESS', 'SUCCESS', 'SUCCESS', 'SUCCESS', 
                'SUCCESS', 'SUCCESS', 'PROCESSING', 'FAILED', 'REFUNDED')

            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) a,

            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) c,
            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) d,
            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) e,
            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4 UNION SELECT 5) f,
            (SELECT 1 UNION SELECT 2 UNION SELECT 3 UNION SELECT 4) g,
            (SELECT 1) r
        LIMIT 50000;

        -- Insert Core Account Log (Simulate 98% match rate)
        SET @log_num = 0; 
        
        INSERT INTO dwd_core_account_log (log_id, order_no, account_amount, book_time, status)
        SELECT 
            CONCAT('L_NORM_', LPAD(@log_num := @log_num + 1, 10, '0')),
            order_no,
            trans_amount,
            DATE_ADD(trans_time, INTERVAL FLOOR(RAND() * 5) SECOND),
            1
        FROM dwd_pay_channel_flow
        WHERE flow_id LIKE 'F_NORM_%' 
          AND RAND() > 0.02
        ORDER BY flow_id DESC 
        LIMIT 50000;

        COMMIT;
        
        SET v_current_batch = v_current_batch + 1;
        
    END WHILE;

    SELECT 'Data Generation Completed Successfully!' AS Result, 
           CONCAT('Total Normal Rows: ~', v_total_normal_rows) AS Info,
           'Skew Rows: 100000 vs 99999' AS Info2;

END$$
